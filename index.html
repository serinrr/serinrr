自动进入页面书签脚本

点击下方按钮复制脚本，拖动到浏览器书签栏

javascript:(function(){var buttons=document.querySelectorAll('button,[role="button"]');for(var i=0;i<buttons.length;i++){var text=(buttons[i].textContent||buttons[i].innerText||'').trim();if(text.includes('预览')||text.includes('进入')||text.includes('Preview')||text.includes('Enter')){buttons[i].click();return;}}console.log('未找到按钮');alert('未找到预览按钮，请等待页面完全加载后再执行！');})();
